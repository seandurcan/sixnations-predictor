generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

model EmailVerification {
  id         Int      @id @default(autoincrement())
  userId     Int
  token      String   @unique
  expiresAt  DateTime
  verifiedAt DateTime?

  user User @relation(fields: [userId], references: [id], onDelete: Cascade)

  @@index([token])
  @@index([userId])
}


model User {
  id                    Int                    @id @default(autoincrement())
  firstName             String
  lastName              String
  email                 String                 @unique
  mobile                String
  passwordHash          String
  role                  UserRole               @default(USER)
  paymentStatus         PaymentStatus          @default(PENDING)
  themePreference       ThemePreference        @default(SYSTEM)
  emailVerified         Boolean                @default(false)
  paidAt                DateTime?
  totalPoints           Int                    @default(0)
  cumulativeError       Int                    @default(0)
  exactScores           Int                    @default(0)
  registrationOrder     Int
  predictionsSubmitted  Boolean                @default(false)
  predictionSubmittedAt DateTime?
  createdAt             DateTime               @default(now())
  updatedAt             DateTime               @updatedAt
  deletedAt             DateTime?
  snapshots             LeaderboardSnapshot[]
  passwordResets        PasswordReset[]
  payments              Payment[]
  predictions           Prediction[]
  submissions           PredictionSubmission[]
  sessions 		Session[]
  emailVerifications 	EmailVerification[]

  @@index([email])
  @@index([paymentStatus])
}

model Team {
  id              Int @id @default(autoincrement())

  name            String @unique

  shortCode       String @unique

  country         String

  flagSvg         String

  primaryColor    String

  secondaryColor  String?

  homeVenue       String?

  city            String?

  foundedYear     Int?

  worldRanking    Int?

  createdAt       DateTime @default(now())

  updatedAt       DateTime @updatedAt

  homeMatches     Match[] @relation("HomeTeam")

  awayMatches     Match[] @relation("AwayTeam")
}
model Tournament {
  id               Int                    @id @default(autoincrement())
  year             Int                    @unique
  name             String
  status           TournamentStatus
  firstKickoff     DateTime
  predictionLockAt DateTime
  createdAt        DateTime               @default(now())
  updatedAt        DateTime               @updatedAt
  snapshots        LeaderboardSnapshot[]
  matches          Match[]
  submissions      PredictionSubmission[]
}

model Match {
  id                Int @id @default(autoincrement())

  tournamentId      Int

  round             Int
  matchNumber       Int

  kickoffTime       DateTime

  venue             String

  city              String?
  country           String?

  stadiumCapacity   Int?

  referee           String?

  attendance        Int?

  weather           String?

  homeTeamId        Int

  awayTeamId        Int

  actualHomeScore   Int?

  actualAwayScore   Int?

  homeTries         Int?

  awayTries         Int?

  completed         Boolean @default(false)

  sourceUrl         String?

  notes             String?

  createdAt         DateTime @default(now())

  updatedAt         DateTime @updatedAt

  tournament Tournament @relation(fields: [tournamentId], references: [id])

  homeTeam Team @relation("HomeTeam", fields: [homeTeamId], references: [id])

  awayTeam Team @relation("AwayTeam", fields: [awayTeamId], references: [id])

  predictions Prediction[]

  @@index([tournamentId])

  @@index([kickoffTime])

@@unique([tournamentId, matchNumber])
}

model Prediction {
  id                 Int      @id @default(autoincrement())

  userId             Int
  matchId            Int

  predictedHomeScore Int
  predictedAwayScore Int

  pointsAwarded      Int      @default(0)

  errorValue         Int      @default(0)

  differenceScore    Int      @default(0)

  exactScore         Boolean  @default(false)

  createdAt          DateTime @default(now())
  updatedAt          DateTime @updatedAt

  match              Match    @relation(fields: [matchId], references: [id], onDelete: Cascade)

  user               User     @relation(fields: [userId], references: [id], onDelete: Cascade)

  @@unique([userId, matchId])

  @@index([userId])

  @@index([matchId])
}

model Payment {
  id                Int             @id @default(autoincrement())
  userId            Int
  provider          PaymentProvider
  providerPaymentId String          @unique
  amount            Decimal         @db.Decimal(10, 2)
  currency          String          @default("EUR")
  status            PaymentStatus
  completedAt       DateTime?
  createdAt         DateTime        @default(now())
  updatedAt         DateTime        @updatedAt
  user              User            @relation(fields: [userId], references: [id], onDelete: Cascade)

  @@index([userId])
  @@index([status])
}

model PasswordReset {
  id        Int      @id @default(autoincrement())
  userId    Int
  token     String   @unique
  expiresAt DateTime
  used      Boolean  @default(false)
  createdAt DateTime @default(now())
  user      User     @relation(fields: [userId], references: [id], onDelete: Cascade)
}

model PredictionSubmission {
  id               Int        @id @default(autoincrement())
  userId           Int
  tournamentId     Int
  receiptReference String     @unique
  submittedAt      DateTime   @default(now())
  ipAddress        String?
  userAgent        String?
  tournament       Tournament @relation(fields: [tournamentId], references: [id])
  user             User       @relation(fields: [userId], references: [id])
}

model LeaderboardSnapshot {
  id              Int        @id @default(autoincrement())
  tournamentId    Int
  userId          Int
  snapshotNumber  Int
  rank            Int
  previousRank    Int?
  rankMovement    Int?
  totalPoints     Int
  cumulativeError Int
  exactScores     Int
  generatedAt     DateTime   @default(now())
  tournament      Tournament @relation(fields: [tournamentId], references: [id])
  user            User       @relation(fields: [userId], references: [id])

  @@unique([tournamentId, snapshotNumber, userId])
}

model ScoreAudit {
  id           Int      @id @default(autoincrement())
  matchId      Int
  previousHome Int?
  previousAway Int?
  newHome      Int
  newAway      Int
  adminUserId  Int
  createdAt    DateTime @default(now())

  @@index([matchId])
}

model TournamentWinner {
  id           Int      @id @default(autoincrement())
  tournamentId Int      @unique
  userId       Int
  finalPoints  Int
  createdAt    DateTime @default(now())
}

model SystemSetting {
  id        Int      @id @default(autoincrement())
  key       String   @unique
  value     String
  updatedAt DateTime @updatedAt
}

enum UserRole {
  USER
  ADMIN
}

enum PaymentStatus {
  PENDING
  COMPLETED
  FAILED
  REFUNDED
}

enum TournamentStatus {
  DRAFT
  OPEN
  LOCKED
  COMPLETED
}

enum PaymentProvider {
  STRIPE
  PAYPAL
  TEST
}

enum ThemePreference {
  LIGHT
  DARK
  SYSTEM
}

model Session {
  id        Int      @id @default(autoincrement())
  token     String   @unique
  userId    Int
  expiresAt DateTime
  createdAt DateTime @default(now())

  user User @relation(fields: [userId], references: [id], onDelete: Cascade)

  @@index([token])
  @@index([userId])
}
